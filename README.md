# Chọn 1 trong 2 cách: SSH hoặc HTTPS

# (A) SSH (khuyến nghị)
git remote add origin git@github.com:<username-or-org>/<repo>.git

# (B) HTTPS
git remote add origin https://github.com/<username-or-org>/<repo>.git

# Push lần đầu & đặt upstream
git push -u origin main
